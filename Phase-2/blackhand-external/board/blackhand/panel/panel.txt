# ─────────────────────────────────────────────────────────────────────────────
# BlackHand OS — ST7789P init sequence
# Waveshare 1.83inch Touch LCD Module · 240x284 IPS · 4-wire SPI
#
# Compiled to panel.bin by mkpanelbin.py, read by the kernel's
# panel-mipi-dbi driver from /lib/firmware/panel.bin at probe time.
#
# PROVENANCE
# ----------
# Values are taken from Waveshare's own ST7789 fbtft driver, HSD20_IPS branch:
#   https://github.com/waveshareteam/LCD-show/blob/master/st7789_module/fb_st7789v.c
# Waveshare publishes no Raspberry Pi guide for this module (ESP32 and Arduino
# only), so this is the closest thing to a vendor-sanctioned sequence. If the
# panel comes up with wrong gamma or colour, re-derive from LCD_Init() in the
# vendor demo ZIP and regenerate — that is the intended tuning loop.
#
# NOT INCLUDED HERE (the driver does these itself, do not duplicate):
#   - hardware reset       → reset-gpio in config.txt
#   - CASET/RASET windowing → x-offset / y-offset dtparams
#   - backlight enable     → backlight-gpio in config.txt
#
# SYNTAX
#   command <hex> [<hex> ...]   one command, zero or more parameter bytes
#   delay   <ms>                1..255 ms (encoded as MIPI NOP + 1 param byte)
#   # ...                       comment
# ─────────────────────────────────────────────────────────────────────────────

command 0x11                                     # SLPOUT — leave sleep mode
delay   120                                      # datasheet: 120 ms before next cmd

command 0x3A 0x05                                # COLMOD — 16 bpp RGB565
command 0x36 0x00                                # MADCTL — portrait, RGB order
                                                 #   bit3 = BGR  (set 0x08 if R/B swap)
                                                 #   bit6 = MX, bit7 = MY (flip)

command 0xB2 0x05 0x05 0x00 0x33 0x33            # PORCTRL  — porch, IPS values
command 0xB7 0x75                                # GCTRL    — VGH 13.26V / VGL -10.43V
command 0xC2 0x01 0xFF                           # VDVVRHEN — VDV/VRH from command, not NVM
command 0xC3 0x13                                # VRHS     — VAP/VAN
command 0xC4 0x20                                # VDVS     — VDV = 0V
command 0xBB 0x22                                # VCOMS    — VCOM = 0.9V
command 0xC5 0x20                                # VCMOFSET — VCOM offset = 0V
command 0xD0 0xA4 0xA1                           # PWCTRL1  — AVDD 6.8V / AVCL -4.8V / VDS 2.3V

# Gamma — HSD20 IPS panel curves
command 0xE0 0xD0 0x05 0x0A 0x09 0x08 0x05 0x2E 0x44 0x45 0x0F 0x17 0x16 0x2B 0x33   # PVGAMCTRL
command 0xE1 0xD0 0x05 0x0A 0x09 0x08 0x05 0x2E 0x43 0x45 0x0F 0x16 0x16 0x2B 0x33   # NVGAMCTRL

command 0x21                                     # INVON — REQUIRED on this IPS panel.
                                                 # Omit it and every colour inverts.
command 0x13                                     # NORON — normal display mode
command 0x29                                     # DISPON
delay   100
