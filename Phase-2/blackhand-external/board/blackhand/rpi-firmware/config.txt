# Black Hand OS — HyperPixel 4.0
#
# Display: firmware-driven DPI (not KMS — avoids circular dep on kernel 6.6)
# Touch:   i2c-gpio overlay for Goodix GT911 on pins 10/11
# Panel init: hyperpixel4-init userspace tool (SPI bit-bang)

arm_64bit=1
kernel=Image
gpu_mem=256

# ── Firmware-driven DPI — exact Pimoroni Pi 4 values ──
# Panel is natively 480x800 (portrait).
enable_dpi_lcd=1
dpi_group=2
dpi_mode=87
dpi_output_format=0x7f216
dpi_timings=480 0 10 16 59 800 0 15 113 15 0 0 0 60 0 32000000 6

# Match the native panel resolution (portrait)
framebuffer_width=480
framebuffer_height=800

# Rotate 180 degrees (flip both axes)
display_rotate=2

# NO vc4 overlay — let firmware own the display entirely.

# enable_uart=1 fixes the VPU core_freq to 250 MHz so the mini-UART baud
# rate is stable. It does NOT swap the UARTs on Pi 4 — PL011 (ttyAMA0)
# stays on Bluetooth. Without this, mini-UART baud drifts at load.
#
# krnbt=on is intentionally ABSENT. With krnbt=on the VideoCore firmware
# skips its own UART initialisation and hands the BCM43455 directly to
# the Linux serdev driver. That changes the UART clock-switch sequence and
# causes "command 0x0a0a tx timeout" (-110 ETIMEDOUT) during firmware
# patch upload — the chip never gets its UART reconfigured to 3 Mbaud.
# Without krnbt, the VC firmware does the UART setup correctly first.
# The RPi downstream DT already has the bcm43438-bt serdev node built in,
# so hci0 still appears automatically without the krnbt flag.
enable_uart=1

# ── I2C-GPIO bus for touchscreen ──
# The Goodix GT911 touch controller sits on GPIO 10 (SDA) / GPIO 11 (SCL).
# We create a bit-banged I2C bus (bus 3) using the i2c-gpio overlay.
dtoverlay=i2c-gpio,bus=3,i2c_gpio_sda=10,i2c_gpio_scl=11,i2c_gpio_delay_us=4

# ── Goodix touchscreen on the I2C-GPIO bus ──
# Custom overlay that defines the GT911 at addresses 0x14 and 0x5d
# with interrupt on GPIO 27.
dtoverlay=hyperpixel4-touch

# Enable onboard audio (headphone jack / HDMI)
dtparam=audio=on

# ── Silence cosmetic dmesg warnings (UART pinctrl + BT regulator) ──
# Declares what the Pi firmware already configured at power-on so Linux
# stops logging "no valid maps for state default" and "supply not found".
dtoverlay=blackhand-fixes

disable_splash=1
boot_delay=0
